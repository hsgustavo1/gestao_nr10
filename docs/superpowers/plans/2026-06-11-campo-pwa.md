# Campo PWA Offline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Criar um Progressive Web App separado em `campo-pwa/` que permite inspetores coletar pontos de campo (fotos + modos de falha) completamente offline, sincronizando com Supabase ao reconectar.

**Architecture:** App Vite + React 19 standalone em `campo-pwa/` no mesmo repositório. Usa o mesmo projeto Supabase (`@supabase/supabase-js`) com as mesmas credenciais. Armazenamento local via Dexie.js (IndexedDB). Service worker via `vite-plugin-pwa` (Workbox). Nenhuma alteração no app principal.

**Tech Stack:** Vite 6, React 19, TypeScript, Dexie.js 4, @supabase/supabase-js 2, vite-plugin-pwa, React Router DOM v7, shadcn/ui (subset), Tailwind CSS v4

```ts
import Dexie, { type Table } from 'dexie'
import type {
  FieldInspection, FieldNode, FieldPoint, FieldFinding, FieldPhoto, RtiModoFalha,
} from '@/lib/types'

export type SyncOperation = 'insert' | 'update' | 'delete'

export interface SyncQueueItem {
  id: string
  operation: SyncOperation
  table: 'inspections' | 'nodes' | 'points' | 'findings' | 'photos'
  payload: unknown
  local_id: string
  remote_id: string | null
  attempts: number
  created_at: string
}

export type LocalInspection = FieldInspection & { _synced: boolean }
export type LocalNode = FieldNode & { _synced: boolean }
export type LocalPoint = FieldPoint & { _synced: boolean }
export type LocalFinding = FieldFinding & { _synced: boolean }
export type LocalPhoto = Omit<FieldPhoto, 'file_path'> & {
  file_path: string | null
  blob: Blob | null
  _synced: boolean
}
export type LocalModoFalha = RtiModoFalha

class CampoDatabase extends Dexie {
  inspections!: Table<LocalInspection>
  nodes!: Table<LocalNode>
  points!: Table<LocalPoint>
  findings!: Table<LocalFinding>
  photos!: Table<LocalPhoto>
  modos_falha!: Table<LocalModoFalha>
  sync_queue!: Table<SyncQueueItem>

  constructor() {
    super('campo-nr10')
    this.version(1).stores({
      inspections: 'id, _synced, status, responsavel_id',
      nodes: 'id, inspection_id, parent_id, _synced',
      points: 'id, inspection_id, node_id, _synced',
      findings: 'id, point_id, _synced',
      photos: 'id, point_id, _synced',
      modos_falha: 'id, categoria',
      sync_queue: '++id, created_at, attempts, table',
    })
  }
}

export const db = new CampoDatabase()
```

```ts
import type { LocalNode } from '@/db/dexie'
import type { NodeType } from '@/lib/types'

export function nodePath(nodeId: string, allNodes: LocalNode[]): LocalNode[] {
  const nodeMap = new Map(allNodes.map((n) => [n.id, n]))
  const path: LocalNode[] = []
  let current = nodeMap.get(nodeId)
  while (current) {
    path.unshift(current)
    current = current.parent_id ? nodeMap.get(current.parent_id) : undefined
  }
  return path
}

export function filhosDoNo(parentId: string | null, allNodes: LocalNode[]): LocalNode[] {
  return allNodes
    .filter((n) => n.parent_id === parentId)
    .sort((a, b) => a.ordem - b.ordem)
}

export function proximoNivel(tipo: NodeType | null): NodeType | null {
  if (tipo === null) return 'setor'
  if (tipo === 'setor') return 'ativo'
  if (tipo === 'ativo') return 'componente'
  return null
}

export function labelDoTipo(tipo: NodeType | null): string {
  if (tipo === 'setor') return 'Setor'
  if (tipo === 'ativo') return 'Ativo'
  if (tipo === 'componente') return 'Componente'
  return 'Item'
}
```

```ts
import { v4 as uuid } from 'uuid'
import { supabase } from '@/lib/supabase'
import { db } from '@/db/dexie'
import type { SyncOperation } from '@/db/dexie'

// ── Download: Supabase → Dexie ────────────────────────────────────────────────

export async function downloadAll(userId: string): Promise<void> {
  await downloadModosFalha()
  await downloadInspections(userId)
}

async function downloadModosFalha(): Promise<void> {
  const { data, error } = await supabase.from('rti_modos_falha').select('*')
  if (error) throw error
  await db.modos_falha.bulkPut(data ?? [])
}

async function downloadInspections(userId: string): Promise<void> {
  const { data: inspections, error } = await supabase
    .from('field_inspections')
    .select('*')
    .eq('responsavel_id', userId)
  if (error) throw error

  const rows = (inspections ?? []).map((i) => ({ ...i, _synced: true as const }))
  await db.inspections.bulkPut(rows)

  const activeIds = rows
    .filter((i) => i.status === 'em_andamento')
    .map((i) => i.id)

  for (const id of activeIds) {
    await downloadInspectionData(id)
  }
}

async function downloadInspectionData(inspectionId: string): Promise<void> {
  const [nodes, points, findings, photos] = await Promise.all([
    supabase.from('field_nodes').select('*').eq('inspection_id', inspectionId),
    supabase.from('field_points').select('*').eq('inspection_id', inspectionId),
    supabase.from('field_findings').select('*').in(
      'point_id',
      (await supabase.from('field_points').select('id').eq('inspection_id', inspectionId)).data?.map((p) => p.id) ?? [],
    ),
    supabase.from('field_photos').select('*').in(
      'point_id',
      (await supabase.from('field_points').select('id').eq('inspection_id', inspectionId)).data?.map((p) => p.id) ?? [],
    ),
  ])

  if (nodes.data) await db.nodes.bulkPut(nodes.data.map((n) => ({ ...n, _synced: true })))
  if (points.data) await db.points.bulkPut(points.data.map((p) => ({ ...p, _synced: true })))
  if (findings.data) await db.findings.bulkPut(findings.data.map((f) => ({ ...f, _synced: true })))
  if (photos.data) {
    await db.photos.bulkPut(
      photos.data.map((p) => ({ ...p, blob: null, _synced: true })),
    )
  }
}

// ── Enqueue: local → sync_queue ──────────────────────────────────────────────

export function enqueue(
  table: 'inspections' | 'nodes' | 'points' | 'findings' | 'photos',
  operation: SyncOperation,
  payload: unknown,
  localId: string,
): Promise<unknown> {
  return db.sync_queue.add({
    id: uuid(),
    operation,
    table,
    payload,
    local_id: localId,
    remote_id: null,
    attempts: 0,
    created_at: new Date().toISOString(),
  })
}

// ── Upload: sync_queue → Supabase ─────────────────────────────────────────────

export async function processQueue(): Promise<void> {
  const items = await db.sync_queue.orderBy('created_at').filter((i) => i.attempts < 3).toArray()

  for (const item of items) {
    try {
      if (item.table === 'photos') {
        await uploadPhoto(item.local_id)
      } else {
        await uploadRecord(item)
      }
      await db.sync_queue.delete(item.id)
    } catch {
      await db.sync_queue.update(item.id, { attempts: item.attempts + 1 })
      const delay = Math.pow(2, item.attempts) * 1000
      await new Promise((r) => setTimeout(r, delay))
    }
  }
}

async function uploadRecord(item: { operation: SyncOperation; table: string; payload: unknown }): Promise<void> {
  const payload = item.payload as Record<string, unknown>
  if (item.operation === 'insert') {
    const { error } = await supabase.from(item.table).insert(payload)
    if (error) throw error
  } else if (item.operation === 'update') {
    const { id, ...rest } = payload
    const { error } = await supabase.from(item.table).update(rest).eq('id', id)
    if (error) throw error
  } else if (item.operation === 'delete') {
    const { error } = await supabase.from(item.table).delete().eq('id', payload.id)
    if (error) throw error
  }
}

async function uploadPhoto(localId: string): Promise<void> {
  const photo = await db.photos.get(localId)
  if (!photo?.blob) throw new Error(`Blob não encontrado para foto ${localId}`)

  const ext = photo.blob.type.split('/')[1] ?? 'jpg'
  const remotePath = `campo/${localId}.${ext}`

  const { error: storageError } = await supabase.storage
    .from('rti-evidencias')
    .upload(remotePath, photo.blob, { contentType: photo.blob.type, upsert: true })
  if (storageError) throw storageError

  const { error: dbError } = await supabase.from('field_photos').insert({
    id: photo.id,
    point_id: photo.point_id,
    file_path: remotePath,
    descricao: photo.descricao,
    created_at: photo.created_at,
  })
  if (dbError) throw dbError

  await db.photos.update(localId, { file_path: remotePath, _synced: true })
}

// ── Connectivity watcher ──────────────────────────────────────────────────────

export function startConnectivityWatcher(): () => void {
  const handler = () => {
    if (navigator.onLine) {
      processQueue().catch(console.error)
    }
  }
  window.addEventListener('online', handler)
  if (navigator.onLine) handler()
  return () => window.removeEventListener('online', handler)
}
```

```tsx
        const findingId = uuid()
        const finding = {
          id: findingId,
          point_id: pointId,
          modo_falha_id: modoId,
          observacao: null,
          severidade: null,
          created_at: now,
          _synced: false,
        }
        await db.findings.add(finding)
        await enqueue('findings', 'insert', finding, findingId)
      }

      navigate(-1)
    } finally {
      setSaving(false)
    }
  }

  return (
    <div className="flex flex-col min-h-full pb-safe">
      <header className="flex items-center gap-2 px-4 py-3 border-b border-slate-800">
        <button onClick={() => navigate(-1)} className="p-1.5 rounded-lg hover:bg-slate-800">
          <ArrowLeft className="h-4 w-4" />
        </button>
        <h1 className="font-semibold">Novo ponto de coleta</h1>
      </header>

      <div className="flex-1 overflow-y-auto p-4 space-y-6">
        {/* Fotos */}
        <section className="space-y-3">
          <h2 className="text-sm font-semibold text-slate-300 uppercase tracking-wide">Fotos</h2>

          {/* câmera — sem multiple (iOS não suporta capture+multiple) */}
          <input
            ref={cameraRef}
            type="file"
            accept="image/*"
            capture="environment"
            className="hidden"
            onChange={(e) => addFotos(e.target.files)}
          />
          {/* galeria — com multiple */}
          <input
            ref={galeriaRef}
            type="file"
            accept="image/*"
            multiple
            className="hidden"
            onChange={(e) => addFotos(e.target.files)}
          />

          <div className="flex gap-2">
            <button
              onClick={() => cameraRef.current?.click()}
              className="flex items-center gap-2 rounded-lg border border-slate-700 px-4 py-2.5 text-sm hover:bg-slate-800 transition-colors"
            >
              <Camera className="h-4 w-4" />
              Tirar foto
            </button>
            <button
              onClick={() => galeriaRef.current?.click()}
              className="flex items-center gap-2 rounded-lg border border-slate-700 px-4 py-2.5 text-sm hover:bg-slate-800 transition-colors"
            >
              <ImagePlus className="h-4 w-4" />
              Galeria
            </button>
          </div>

          {fotos.length > 0 && (
            <div className="grid grid-cols-3 gap-2">
              {fotos.map((foto) => (
                <div key={foto.localId} className="relative aspect-square">
                  <img
                    src={foto.url}
                    alt="foto capturada"
                    className="h-full w-full rounded-lg object-cover"
                  />
                  <button
                    onClick={() => removeFoto(foto.localId)}
                    className="absolute top-1 right-1 rounded-full bg-black/60 p-1"
                  >
                    <X className="h-3 w-3" />
                  </button>
                </div>
              ))}
            </div>
          )}
        </section>

        {/* Modos de falha */}
        <section className="space-y-3">
          <h2 className="text-sm font-semibold text-slate-300 uppercase tracking-wide">
            Modos de falha
          </h2>
          {categorias.map((cat) => (
            <details key={cat} className="rounded-xl bg-slate-800 overflow-hidden">
              <summary className="px-4 py-3 cursor-pointer text-sm font-medium list-none flex items-center justify-between">
                {cat}
                <span className="text-xs text-slate-400">
                  {modosFalha?.filter((m) => m.categoria === cat && selectedModos.includes(m.id)).length || ''}
                </span>
              </summary>
              <div className="border-t border-slate-700 divide-y divide-slate-700/50">
                {modosFalha
                  ?.filter((m) => m.categoria === cat)
                  .map((modo) => (
                    <button
                      key={modo.id}
                      onClick={() => toggleModo(modo.id)}
                      className={`w-full flex items-start gap-3 px-4 py-3 text-left text-sm transition-colors ${
                        selectedModos.includes(modo.id) ? 'bg-blue-900/30' : 'hover:bg-slate-700/50'
                      }`}
                    >
                      <div className={`mt-0.5 h-4 w-4 rounded flex items-center justify-center shrink-0 border ${
                        selectedModos.includes(modo.id)
                          ? 'bg-blue-600 border-blue-600'
                          : 'border-slate-500'
                      }`}>
                        {selectedModos.includes(modo.id) && <Check className="h-3 w-3" />}
                      </div>
                      <span>{modo.descricao}</span>
                    </button>
                  ))}
              </div>
            </details>
          ))}
        </section>

        {/* Campos opcionais */}
        <section className="space-y-3">
          <h2 className="text-sm font-semibold text-slate-300 uppercase tracking-wide">
            Informações (opcional)
          </h2>
          <input
            value={titulo}
            onChange={(e) => setTitulo(e.target.value)}
            placeholder="Título do ponto"
            className="w-full rounded-lg bg-slate-800 border border-slate-700 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
          />
          <textarea
            value={observacao}
            onChange={(e) => setObservacao(e.target.value)}
            placeholder="Observação..."
            rows={3}
            className="w-full rounded-lg bg-slate-800 border border-slate-700 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500 resize-none"
          />
        </section>
      </div>

      <div className="p-4 border-t border-slate-800">
        <button
          onClick={handleSave}
          disabled={saving || (fotos.length === 0 && selectedModos.length === 0)}
          className="w-full flex items-center justify-center gap-2 rounded-xl bg-blue-600 hover:bg-blue-500 disabled:opacity-40 disabled:cursor-not-allowed px-4 py-3.5 font-semibold transition-colors"
        >
          {saving ? 'Salvando...' : 'Salvar ponto'}
        </button>
      </div>
    </div>
  )
}
```
